```js
const ACTIONS = Object.freeze({
  poke: Object.freeze({ name: "poke", sequence: Object.freeze(["poke", "recover"]), effect: null, cooldownMs: 450 }),
  slap: Object.freeze({ name: "slap", sequence: Object.freeze(["slap", "recover"]), effect: "slap", cooldownMs: 650 }),
  whip: Object.freeze({ name: "whip", sequence: Object.freeze(["whipHit", "recover"]), effect: "whip", cooldownMs: 900 }),
});

export function getTroyAction(name) {
  return ACTIONS[name] || null;
}

export function isTroyTool(name) {
  return Object.hasOwn(ACTIONS, name);
}

export function isActionReady(lastRunAt, now, cooldownMs) {
  return now - lastRunAt >= cooldownMs;
}

export function createLatestActionQueue() {
  let pending = null;
  return {
    request(name) { pending = isTroyTool(name) ? name : null; },
    take() { const value = pending; pending = null; return value; },
    clear() { pending = null; },
    size() { return pending === null ? 0 : 1; },
  };
}
```

Return plain copies from `getTroyAction` if test mutation shows the frozen internal structure leaking into controller code.

```js
export function pickTroyLine(lines, previousIndex = -1, randomValue = Math.random()) {
  if (!Array.isArray(lines) || lines.length === 0) return { line: "", index: -1 };
  if (lines.length === 1) return { line: lines[0], index: 0 };
  const candidate = Math.min(lines.length - 1, Math.floor(Math.max(0, randomValue) * lines.length));
  const index = candidate === previousIndex ? (candidate + 1) % lines.length : candidate;
  return { line: lines[index], index };
}
```
